library IEEE;
use IEEE.STD_LOGIC_1164.ALL;
use IEEE.NUMERIC_STD.ALL;

-- 16 x 12-bit ROM, Set 9 program (Appendix B)
entity program_rom is
    port (
        addr  : in  std_logic_vector(3 downto 0);
        instr : out std_logic_vector(11 downto 0)
    );
end program_rom;

architecture rtl of program_rom is
    type rom_t is array (0 to 15) of std_logic_vector(11 downto 0);
    constant ROM : rom_t := (
        0      => x"1FF",   -- LDI 0xFF
        1      => x"700",   -- NOT
        2      => x"C05",   -- JZ  0x05
        3      => x"277",   -- ADD 0x77
        4      => x"277",   -- ADD 0x77
        5      => x"211",   -- ADD 0x11
        6      => x"800",   -- SHL
        7      => x"C00",   -- JZ  0x00
        8      => x"800",   -- SHL
        others => x"000"    -- NOP
    );
begin
    instr <= ROM(to_integer(unsigned(addr)));
end rtl;
