library IEEE;
use IEEE.STD_LOGIC_1164.ALL;

-- Truth-table test for all basic gates
entity tb_gates is
end tb_gates;

architecture sim of tb_gates is
    signal a, b : std_logic := '0';
    signal y_and, y_or, y_not, y_nand, y_nor, y_xor, y_xnor : std_logic;
begin
    u_and  : entity work.and_gate  port map (a => a, b => b, y => y_and);
    u_or   : entity work.or_gate   port map (a => a, b => b, y => y_or);
    u_not  : entity work.not_gate  port map (a => a,         y => y_not);
    u_nand : entity work.nand_gate port map (a => a, b => b, y => y_nand);
    u_nor  : entity work.nor_gate  port map (a => a, b => b, y => y_nor);
    u_xor  : entity work.xor_gate  port map (a => a, b => b, y => y_xor);
    u_xnor : entity work.xnor_gate port map (a => a, b => b, y => y_xnor);

    process
        variable errors : integer := 0;
    begin
        for i in 0 to 3 loop
            if (i / 2) = 1 then a <= '1'; else a <= '0'; end if;
            if (i mod 2) = 1 then b <= '1'; else b <= '0'; end if;
            wait for 10 ns;
            if y_and  /= (a and b)  then errors := errors + 1; report "AND fail"  severity error; end if;
            if y_or   /= (a or b)   then errors := errors + 1; report "OR fail"   severity error; end if;
            if y_not  /= (not a)    then errors := errors + 1; report "NOT fail"  severity error; end if;
            if y_nand /= (a nand b) then errors := errors + 1; report "NAND fail" severity error; end if;
            if y_nor  /= (a nor b)  then errors := errors + 1; report "NOR fail"  severity error; end if;
            if y_xor  /= (a xor b)  then errors := errors + 1; report "XOR fail"  severity error; end if;
            if y_xnor /= (a xnor b) then errors := errors + 1; report "XNOR fail" severity error; end if;
        end loop;
        if errors = 0 then
            report "ALL GATE TESTS PASSED" severity note;
        end if;
        wait;
    end process;
end sim;
