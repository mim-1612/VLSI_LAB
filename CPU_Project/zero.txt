library IEEE;
use IEEE.STD_LOGIC_1164.ALL;

-- 8-bit ripple-carry adder (carry out is discarded)
entity adder8 is
    port (
        a, b : in  std_logic_vector(7 downto 0);
        s    : out std_logic_vector(7 downto 0)
    );
end adder8;

architecture structural of adder8 is
    signal c : std_logic_vector(8 downto 0);
begin
    c(0) <= '0';
    gen_fa : for i in 0 to 7 generate
        fa : entity work.full_adder
            port map (a => a(i), b => b(i), cin => c(i),
                      sum => s(i), cout => c(i+1));
    end generate;
end structural;
