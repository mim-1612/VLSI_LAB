library IEEE;
use IEEE.STD_LOGIC_1164.ALL;

entity tb_accumulator4 is
end tb_accumulator4;

architecture behavior of tb_accumulator4 is
component accumulator4
Port ( A : in STD_LOGIC_VECTOR (3 downto 0);
B : in STD_LOGIC_VECTOR (3 downto 0);
CLK : in STD_LOGIC;
RESET : in STD_LOGIC;
Q : out STD_LOGIC_VECTOR (3 downto 0));
end component;

signal A, B, Q : STD_LOGIC_VECTOR (3 downto 0) := "0000";
signal CLK : STD_LOGIC := '0';
signal RESET : STD_LOGIC := '1';
begin
uut: accumulator4 port map (A => A, B => B, CLK => CLK, RESET => RESET, Q => Q);

clk_process: process
begin
CLK <= '0'; wait for 5 ns;
CLK <= '1'; wait for 5 ns;
end process;

stim: process
begin
RESET <= '1'; A <= "0000"; B <= "0000"; wait for 10 ns;
RESET <= '0'; A <= "0011"; B <= "0101"; wait for 10 ns;
RESET <= '0'; A <= "0010"; B <= "0001"; wait for 10 ns;
RESET <= '0'; A <= "1111"; B <= "0001"; wait for 10 ns;
RESET <= '0'; A <= "1010"; B <= "0101"; wait for 10 ns;
wait;
end process;
end behavior;
